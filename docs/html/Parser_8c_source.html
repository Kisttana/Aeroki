62/* getNextToken - advances and updates parser.ctx */
63void getNextToken(void) {
64 if (!parser.tokens) {
65 fprintf(stderr, "getNextToken: parser not initialized\n");
66 exit(EXIT_FAILURE);
67 }
68 if (parser.current_idx + 1 >= parser.tokens->size) {
69 /* set EOF-like token */
70 parser.current_idx = parser.tokens->size;
71 parser.ctx.type = TOKEN_EOF; /* adapt if necessary */
72 free(parser.ctx.val);
73 parser.ctx.val = strdup("<EOF>");
74
75 dbg("%s","Get Next Token" , "the token is end");
76
77 return;
78 }
79 parser.current_idx++;

81 parser.ctx.type = t._Type;
82 free(parser.ctx.val);
83 parser.ctx.val = strdup(t._Value);
84
85 dbg("%s [%zu], %s, %s ", "Get Next Token" ,
86 "set token context to",
87 parser.current_idx,
88 typeof(parser.ctx.type),
89 parser.ctx.val);

339/* binary RHS parsing (classic precedence climbing) */
340static AST *parseBinOpRHSInternal(int exprPrec, AST *lhs) {
341 while (1) {
342
343 int tokPrec = getTokenPrec(curTokenType());
344 if (tokPrec < exprPrec) return lhs;
345
346 /* operator */
347 char opbuf[8];
348 strncpy(opbuf, curTokenVal(), sizeof(opbuf)-1);
349 opbuf[sizeof(opbuf)-1] = '\0';
350 getNextToken(); /* eat operator */
351
352 AST *rhs = parseUnaryInternal();
353 if (!rhs) {
354 freeAST(lhs);
355 return NULL;
356 }
357
358 /* If next operator has higher precedence, it binds to rhs */
359 int nextPrec = getTokenPrec(curTokenType());
360 if (tokPrec < nextPrec) {
361 rhs = parseBinOpRHSInternal(tokPrec + 1, rhs);
362 if (!rhs) {
363 freeAST(lhs);
364 return NULL;
365 }
366 }
367
368 lhs = newBinaryNode(opbuf, lhs, rhs);
369 }
370}

496void printAST(const AST *node, int depth) {
497 if (!node) {
498 indent_print(depth); printf("(NULL)\n"); return;
499 }
500 switch (node->_Type) {
501 case AST_NUMBER:
502 indent_print(depth); printf("(NUM %Lf)\n", node->_Number);
503 break;
504 case AST_IDENTIFIER:
505 indent_print(depth); printf("(IDENT %s)\n", node->Variable_name);
506 break;
507 case AST_UNARY:
508 indent_print(depth); printf("(UNARY %s\n", node->Binary.op);
509 printAST(node->Binary.right, depth+2);
510 indent_print(depth); printf(")\n");
511 break;
512 case AST_BINARY:
513 indent_print(depth); printf("(BIN %s\n", node->Binary.op);
514 printAST(node->Binary.left, depth+2);
515 printAST(node->Binary.right, depth+2);
516 indent_print(depth); printf(")\n");
517 break;
518 case AST_CALL:
519 indent_print(depth); printf("(CALL %s\n", node->Call.Callee);
520 for (size_t i=0;i<node->Call.Args->size;++i) printAST(vec_at(node->Call.Args, AST*, i), depth+2);
521 indent_print(depth); printf(")\n");
522 break;
523 case AST_PROTOTYPE:
524 indent_print(depth); printf("(PROTO %s (", node->Function.proto->fnName);
525 for (size_t i=0;i<node->Function.proto->Args->size;++i) {
526 printf("%s", vec_at(node->Function.proto->Args, char*, i));
527 if (i+1 < node->Function.proto->Args->size) printf(", ");
528 }
529 printf("))\n");
530 break;
531 case AST_FUNCTION:
532 indent_print(depth); printf("(FUNC %s\n", node->Function.proto->fnName);
533 printAST(node->Function.body, depth+2);
534 indent_print(depth); printf(")\n");

536 default:
537 indent_print(depth); printf("(UNKNOWN)\n");
538 }
